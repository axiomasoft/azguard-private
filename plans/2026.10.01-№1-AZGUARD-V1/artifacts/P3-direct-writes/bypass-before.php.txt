<?php
use AzGuard\Tests\TestCase;
use AzGuard\Storage\StorageRegistry;
use AzGuard\Storage\Schema\StorageSchema;
use AzGuard\Storage\StorageMutation;
use AzGuard\Tests\Fixtures\Storage\SchemaAssertions;
uses(TestCase::class);
it('must reject forwarded Laravel writes before changing grants or panel state', function (): void {
 $storage=app(StorageRegistry::class)->get('default');$schema=app(StorageSchema::class);$schema->drop('default');$schema->create('default');
 try {
  $storage->mutate('admin',function(StorageMutation $m):void {$m->table('role_grants')->insert(SchemaAssertions::grant());$m->touch('admin');});
  $builder=$storage->model('role_grant')->newQuery();
  $method=env('DB_CONNECTION','sqlite')==='pgsql'?'updateFrom':'insertOrIgnoreReturning';
  $args=$method==='updateFrom'?[['origin'=>'import']]:[SchemaAssertions::grant()+['origin'=>'import']];
  $builder->$method(...$args);
  expect([$storage->table('role_grants')->where('origin','import')->count(),$storage->state('admin')->version])->toBe([0,1]);
 } finally { $schema->drop('default'); }
});
